@dataclass(slots=True)
class EMBasedReductionStrategy:
    """Reduce a mixture by sampling it and fitting a configured EM estimator.

    Parameters
    ----------
    estimator : GaussianMixtureEstimator
        Configured estimator to fit. Its ``n_components`` value is the target
        component count. Fitting mutates this estimator, allowing callers to
        inspect it afterward and making ``warm_start`` meaningful when the
        strategy is reused.
    n_samples : int, optional
        Number of samples to draw from the input mixture before fitting. If
        omitted, it is computed from the number of free parameters in the
        reduced full-covariance mixture.
    sample_seed : int, optional
        Seed used only to sample from the input mixture. The estimator's
        ``seed`` independently controls EM initialization.
    samples_per_parameter : int, default=10
        Multiplier applied to the reduced mixture's free-parameter count when
        ``n_samples`` is omitted.

    Examples
    --------
    >>> import torch
    >>> from gmtorch import GaussianMixture, GaussianMixtureEstimator
    >>> from gmtorch.reduction import EMBasedReductionStrategy
    >>> gm = GaussianMixture(
    ...     n_components=2,
    ...     dim=1,
    ...     weights=torch.tensor([0.5, 0.5]),
    ...     means=torch.tensor([[-1.0], [1.0]]),
    ...     covariances=torch.tensor([[[0.5]], [[0.5]]]),
    ... )
    >>> strategy = EMBasedReductionStrategy(
    ...     estimator=GaussianMixtureEstimator(n_components=1, max_iter=2, seed=0),
    ...     n_samples=16,
    ...     sample_seed=0,
    ... )
    >>> reduced, info = strategy.reduce(gm)
    >>> assert reduced.n_components == 1
    >>> assert info.n_iter >= 0
    """

    estimator: GaussianMixtureEstimator
    n_samples: int | None = None
    sample_seed: int | None = None
    samples_per_parameter: int = 10

    @classmethod
    def from_n_components(
        cls,
        n_components: int,
        *,
        n_samples: int | None = None,
        sample_seed: int | None = None,
        samples_per_parameter: int = 10,
    ) -> EMBasedReductionStrategy:
        """Construct a strategy using an estimator with default hyperparameters.

        Parameters
        ----------
        n_components : int
            Target number of components for the reduced mixture.
        n_samples : int, optional
            Explicit number of samples to draw. Automatic sizing is used when omitted.
        sample_seed : int, optional
            Seed used only to sample from the input mixture.
        samples_per_parameter : int, default=10
            Automatic sample-count multiplier used when ``n_samples`` is omitted.

        Returns
        -------
        EMBasedReductionStrategy
            Strategy containing a new, unfitted estimator configured with
            ``n_components`` and otherwise using estimator defaults.

        Examples
        --------
        >>> from gmtorch.reduction import EMBasedReductionStrategy
        >>> strategy = EMBasedReductionStrategy.from_n_components(2, n_samples=32, sample_seed=0)
        >>> assert strategy.target_n_components == 2
        >>> assert strategy.n_samples == 32
        """
        return cls(
            estimator=GaussianMixtureEstimator(n_components=n_components),
            n_samples=n_samples,
            sample_seed=sample_seed,
            samples_per_parameter=samples_per_parameter,
        )

    def reduce(self, gm: GaussianMixture) -> tuple[GaussianMixture, EMFitInfo]:
        """Sample ``gm``, fit the estimator, and return its mixture and fit information.

        Examples
        --------
        >>> import torch
        >>> from gmtorch import GaussianMixture, GaussianMixtureEstimator
        >>> from gmtorch.reduction import EMBasedReductionStrategy
        >>> gm = GaussianMixture(
        ...     n_components=2,
        ...     dim=1,
        ...     weights=torch.tensor([0.5, 0.5]),
        ...     means=torch.tensor([[-1.0], [1.0]]),
        ...     covariances=torch.tensor([[[0.5]], [[0.5]]]),
        ... )
        >>> strategy = EMBasedReductionStrategy(
        ...     GaussianMixtureEstimator(n_components=1, max_iter=2, seed=0),
        ...     n_samples=16,
        ...     sample_seed=0,
        ... )
        >>> reduced, info = strategy.reduce(gm)
        >>> assert reduced.n_components == 1
        >>> assert info.lower_bound.ndim == 0
        """
        n_samples = self._resolve_n_samples(gm)
        samples = gm.sample(n_samples, seed=self.sample_seed)
        self.estimator.fit(samples)

        if self.estimator.fit_info is None:
            msg = "EM reduction completed without fit metadata"
            raise RuntimeError(msg)

        return self.estimator.to_mixture(), self.estimator.fit_info



    def _resolve_n_samples(self, gm: GaussianMixture) -> int:
        """Return the explicit or automatically computed sample count."""
        if self.n_samples is not None:
            self._validate_n_samples(self.n_samples)
            return self.n_samples

        if (
            not isinstance(self.samples_per_parameter, int)
            or isinstance(self.samples_per_parameter, bool)
            or self.samples_per_parameter <= 0
        ):
            msg = "samples_per_parameter must be a positive integer"
            raise ValueError(msg)

        n_parameters = _full_covariance_parameter_count(self.target_n_components, gm.dim)
        n_samples = self.samples_per_parameter * n_parameters
        return max(n_samples, 2, self.target_n_components)

    def _validate_n_samples(self, n_samples: int) -> None:
        """Validate an explicit sample count against estimator requirements."""
        if not isinstance(n_samples, int) or isinstance(n_samples, bool) or n_samples <= 0:
            msg = "n_samples must be a positive integer"
            raise ValueError(msg)

        minimum_samples = max(2, self.target_n_components)
        if n_samples < minimum_samples:
            msg = f"n_samples must be at least {minimum_samples}"
            raise ValueError(msg)




def _full_covariance_parameter_count(n_components: int, dim: int) -> int:
    """Count free parameters in a full-covariance Gaussian mixture."""
    weight_parameters = n_components - 1
    mean_parameters = n_components * dim
    covariance_parameters = n_components * dim * (dim + 1) // 2
    return weight_parameters + mean_parameters + covariance_parameters
