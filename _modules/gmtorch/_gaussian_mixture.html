class GaussianMixture:
    """
    Gaussian Mixture Class.

    Parameters (6):
    - n_components (int)
    - dim (int)
    - weights (Tensor) shape (n_components,)
    - means (Tensor) shape (n_components, dim)
    - covariances (Tensor) shape (n_components, dim, dim)
    - device (torch.device or str)
    """

    def __init__(  # noqa: PLR0913
        self,
        n_components: int = 1,
        dim: int = 1,
        weights: Tensor | None = None,
        means: Tensor | None = None,
        covariances: Tensor | None = None,
        device: torch.device | str = "cpu",
    ) -> None:
        self.device = torch.device(device)

        if weights is None:
            weights = torch.full((n_components,), 1.0 / n_components, device=self.device)
        if means is None:
            means = torch.zeros((n_components, dim), device=self.device)
        if covariances is None:
            covariances = torch.stack(
                [torch.eye(dim, device=self.device) for _ in range(n_components)]
            )

        self._replace_parameters(n_components, dim, weights, means, covariances)

    @classmethod
    def random(  # noqa: PLR0913
        cls,
        n_components: int,
        dim: int,
        pi_alpha: Tensor | None = None,
        mu_rng: tuple[float, float] = (-1.0, 1.0),
        var_df: float | None = None,
        var_scale: Tensor | None = None,
        seed: int | None = None,
        device: torch.device | str = "cpu",
    ) -> "GaussianMixture":
        """
        Generate a random Gaussian Mixture Model.

        Parameters
        ----------
        - n_components (int): number of Gaussian components
        - dim (int): data dimension
        - pi_alpha (Tensor, optional): concentration of Dirichlet prior over weights
        - mu_rng (tuple[float, float]): mean sampling range as (min, max)
        - var_df (float, optional): degrees of freedom for Wishart sampling
        - var_scale (Tensor, optional): covariance matrix used as Wishart scale
        - seed (int, optional): random seed
        - device (torch.device or str): target device

        Returns
        -------
        - GaussianMixture: randomly sampled Gaussian mixture
        """
        if len(mu_rng) != 2 or mu_rng[0] > mu_rng[1]:  # noqa: PLR2004
            msg = f"mu_rng of (min, max) is expected, but got {mu_rng}"
            raise ValueError(msg)

        gm_device = torch.device(device)
        if seed is not None:
            torch.manual_seed(seed)

        if pi_alpha is None:
            pi_alpha = torch.ones(n_components, device=gm_device)
        else:
            pi_alpha = pi_alpha.to(gm_device)
        weights = Dirichlet(pi_alpha).sample()

        means = torch.rand((n_components, dim), device=gm_device)
        means = means * (mu_rng[1] - mu_rng[0]) + mu_rng[0]

        if var_df is None:
            var_df = float(dim + 1)
        if var_scale is None:
            var_scale = torch.eye(dim, device=gm_device)
        else:
            var_scale = var_scale.to(gm_device)
        wishart_dist = Wishart(df=var_df, covariance_matrix=var_scale)
        covariances = wishart_dist.sample(sample_shape=(n_components,))

        return cls(
            n_components=n_components,
            dim=dim,
            weights=weights,
            means=means,
            covariances=covariances,
            device=gm_device,
        )



    @staticmethod
    def _build_distribution(
        weights: Tensor, means: Tensor, covariances: Tensor
    ) -> MixtureSameFamily:
        mixture_distribution = Categorical(probs=weights)
        # `MultivariateNormal` sets event_shape=(dim,), so each feature vector is one event.
        component_distribution = MultivariateNormal(loc=means, covariance_matrix=covariances)
        return MixtureSameFamily(mixture_distribution, component_distribution)

    def _replace_parameters(
        self, n_components: int, dim: int, weights: Tensor, means: Tensor, covariances: Tensor
    ) -> None:
        """Validate and atomically replace all distribution parameters."""
        validated_n_components = self._validate_n_components(n_components)
        validated_dim = self._validate_dim(dim)
        validated_weights = self._validate_weights(weights, validated_n_components)
        validated_means = self._validate_means(means, validated_n_components, validated_dim)
        validated_covariances = self._validate_covariances(
            covariances, validated_n_components, validated_dim
        )
        distribution = self._build_distribution(
            validated_weights, validated_means, validated_covariances
        )

        self.n_components = validated_n_components
        self.dim = validated_dim
        self.weights = validated_weights
        self.means = validated_means
        self.covariances = validated_covariances
        self._distribution = distribution

    def prob_hyperrec(self, event: Tensor, num_samples: int = 100_00) -> Tensor:
        """
        Compute the integral of the Gaussian Mixture over a hyperrectangle known as the "event".

        Parameters
        ----------
        - E (Tensor): hyperrectangle bounds of shape (dim, 2)

        Returns
        -------
        - Tensor: integral value over the hyperrectangle
        """
        prob_hyperrec = torch.zeros(1, device=self.device)
        event = event.to(self.device)
        samples = self.sample(num_samples)
        counter = 0

        def in_hyperrectangle(sample: Tensor, event: Tensor) -> Tensor:
            return torch.all((sample >= event[:, 0]) & (sample <= event[:, 1]))

        in_hyperrectangle_vmap = vmap(lambda sample: in_hyperrectangle(sample, event))
        mask = in_hyperrectangle_vmap(samples)
        counter = mask.sum().item()
        prob_hyperrec[0] = float(counter) / float(num_samples)
        return prob_hyperrec

    def expectation(self) -> Tensor:
        """
        Compute the expectation value of the Gaussian Mixture.

        Returns
        -------
        - Tensor: expectation value of shape (dim)
        """
        expectation = torch.zeros(self.dim, device=self.device, dtype=self.means.dtype)
        for i in range(self.n_components):
            expectation += self.weights[i] * self.means[i, :]
        return expectation

    def covariance(self) -> Tensor:
        """
        Compute the covariance matrix of the Gaussian Mixture.

        Returns
        -------
        - Tensor: covariance matrix of shape (dim, dim)
        """
        covariance = torch.zeros(
            (self.dim, self.dim), device=self.device, dtype=self.covariances.dtype
        )
        for i in range(self.n_components):
            covariance += self.weights[i] * (
                self.covariances[i, :, :]
                + torch.tensordot(self.means[i, :], self.means[i, :], dims=0)
            ) - 1 / self.n_components * torch.tensordot(
                self.expectation(), self.expectation(), dims=0
            )
        return covariance

    def sample(self, n_samples: int, seed: int | None = None) -> Tensor:
        """
        Generate samples from the Gaussian Mixture.

        Parameters
        ----------
        - num_samples (int): number of samples to generate
        - seed (int, optional): random seed for reproducibility

        Returns
        -------
        - Tensor: samples of shape (num_samples, dim)
        """
        if seed is None:
            return self._distribution.sample((n_samples,))  # type: ignore[no-untyped-call,no-any-return]

        # `torch.distributions` sampling APIs do not currently accept a user-supplied
        # Generator, so we fork RNG state to avoid mutating global RNG streams.
        fork_devices = [self.device] if self.device.type == "cuda" else []
        with torch.random.fork_rng(devices=fork_devices):
            torch.manual_seed(seed)
            return self._distribution.sample((n_samples,))  # type: ignore[no-untyped-call,no-any-return]

    def qmc_sampling(
        self, n_samples: int, *, scramble: bool = True, seed: int | None = None
    ) -> Tensor:
        """Generate stratified Sobol samples from this Gaussian mixture.

        Samples are allocated to components with the largest-remainder method,
        so each component receives either the floor or ceiling of its exact
        weighted quota and the total is exactly ``n_samples``. Returned samples
        are grouped in component-index order.

        Parameters
        ----------
        n_samples : int
            Total number of samples to generate.
        scramble : bool, optional
            Whether to apply Owen scrambling independently to each component's
            Sobol sequence.
        seed : int, optional
            Base scrambling seed. Component ``i`` uses ``seed + i``.

        Returns
        -------
        Tensor
            Samples with shape ``(n_samples, dim)`` on this mixture's device.

        Raises
        ------
        ValueError
            If ``n_samples`` or a QMC sampling parameter is invalid.

        Warns
        -----
        UserWarning
            If a positive-weight component is allocated fewer than
            ``MIN_SAMPLE_WARNING`` samples.
        """
        component_counts = _allocate_component_samples(self.weights, n_samples)
        positive_weights = self.weights.detach().to(device="cpu") > 0
        undersampled_components = torch.nonzero(
            positive_weights & (component_counts < MIN_SAMPLE_WARNING), as_tuple=False
        ).flatten()
        if undersampled_components.numel() > 0:
            details = ", ".join(
                f"{component_index} ({component_counts[component_index].item()})"
                for component_index in undersampled_components.tolist()
            )
            warnings.warn(
                f"QMC sample count is below {MIN_SAMPLE_WARNING} for components: {details}",
                UserWarning,
                stacklevel=2,
            )

        component_samples: list[Tensor] = []
        for component_index, component_count in enumerate(component_counts):
            component_n_samples = int(component_count.item())
            if component_n_samples == 0:
                continue
            component_seed = None if seed is None else seed + component_index
            component_samples.append(
                qmc_sample_gaussian(
                    self.means[component_index],
                    self.covariances[component_index],
                    component_n_samples,
                    scramble=scramble,
                    seed=component_seed,
                )
            )
        return torch.cat(component_samples, dim=0)

    def propagate(
        self,
        func: Callable[[Tensor], Tensor],
        n_samples: int,
        target_k: int,
        fit_kwargs: Mapping[str, object] | None = None,
        *,
        sample_seed: int | None = None,
    ) -> "GaussianMixture":
        """Approximate this mixture propagated through ``func``.

        This is a convenience wrapper around ``gmtorch.propagation.propagate_gmm``.
        It samples from this mixture, applies the batch function ``func``, and
        fits a new Gaussian mixture with ``target_k`` components to the
        transformed samples.

        Parameters
        ----------
        func : Callable[[Tensor], Tensor]
            Batch function applied to samples with shape ``(n_samples, self.dim)``.
        n_samples : int
            Number of samples to draw from this mixture.
        target_k : int
            Number of components in the fitted propagated mixture.
        fit_kwargs : Mapping[str, object], optional
            Additional keyword arguments forwarded to ``GaussianMixtureEstimator``.
        sample_seed : int, optional
            Seed used only when sampling from this mixture.

        Returns
        -------
        GaussianMixture
            Gaussian-mixture approximation to ``func(X)``.
        """
        propagation_module = importlib.import_module("gmtorch.propagation")
        propagate_gmm = cast("Callable[..., GaussianMixture]", propagation_module.propagate_gmm)
        return propagate_gmm(self, func, n_samples, target_k, fit_kwargs, sample_seed=sample_seed)

    def scale_(self, scale: float | Tensor) -> "GaussianMixture":
        """Scale this Gaussian mixture in place.

        Each component mean is multiplied by ``scale`` and each component
        covariance is multiplied by ``scale**2``. Component weights are
        preserved.

        Parameters
        ----------
        scale : float or Tensor
            Finite, non-zero real scalar. A tensor scale must be zero-dimensional
            and on the same device as this mixture.

        Returns
        -------
        GaussianMixture
            This updated Gaussian mixture.

        Raises
        ------
        ValueError
            If ``scale`` is not a finite, non-zero real scalar or is on a
            different device.
        """
        if isinstance(scale, Tensor):
            scale_tensor = scale
            if scale_tensor.ndim != 0:
                msg = "scale must be a scalar"
                raise ValueError(msg)
            if scale_tensor.device != self.device:
                msg = "scale and GaussianMixture must be on the same device"
                raise ValueError(msg)
        else:
            scale_tensor = torch.tensor(scale, device=self.device, dtype=self.means.dtype)

        if scale_tensor.is_complex():
            msg = "scale must be real"
            raise ValueError(msg)
        if not torch.isfinite(scale_tensor):
            msg = "scale must be finite"
            raise ValueError(msg)
        if scale_tensor == 0:
            msg = "scale must be non-zero"
            raise ValueError(msg)

        component_dtype = torch.promote_types(self.means.dtype, self.covariances.dtype)
        component_dtype = torch.promote_types(component_dtype, scale_tensor.dtype)
        scale_tensor = scale_tensor.to(dtype=component_dtype)
        means = self.means.to(dtype=component_dtype) * scale_tensor
        covariances = self.covariances.to(dtype=component_dtype) * scale_tensor.square()

        self._replace_parameters(self.n_components, self.dim, self.weights, means, covariances)
        return self

    def shift_(self, offset: Tensor) -> "GaussianMixture":
        """Shift this Gaussian mixture in place.

        The offset is added to every component mean. Component weights and
        covariances are preserved.

        Parameters
        ----------
        offset : Tensor
            Finite real vector with shape ``(self.dim,)`` on the same device as
            this mixture.

        Returns
        -------
        GaussianMixture
            This updated Gaussian mixture.

        Raises
        ------
        ValueError
            If ``offset`` has an invalid shape, contains non-finite or complex
            values, or is on a different device.
        """
        if offset.ndim != 1 or offset.shape[0] != self.dim:
            msg = "offset must have shape (dim)"
            raise ValueError(msg)
        if offset.device != self.device:
            msg = "offset and GaussianMixture must be on the same device"
            raise ValueError(msg)
        if offset.is_complex():
            msg = "offset must be real"
            raise ValueError(msg)
        if not torch.isfinite(offset).all():
            msg = "offset must contain only finite values"
            raise ValueError(msg)

        mean_dtype = torch.promote_types(self.means.dtype, offset.dtype)
        means = self.means.to(dtype=mean_dtype) + offset.to(dtype=mean_dtype)

        self._replace_parameters(self.n_components, self.dim, self.weights, means, self.covariances)
        return self

    def rotate_2d_(self, angle: float | Tensor, center: Tensor | None = None) -> "GaussianMixture":
        """Rotate this two-dimensional Gaussian mixture in place.

        Parameters
        ----------
        angle : float or Tensor
            Finite real scalar angle in radians. Positive angles rotate
            counterclockwise.
        center : Tensor, optional
            Rotation center with shape ``(2,)`` on the same device as this
            mixture. The origin is used by default.

        Returns
        -------
        GaussianMixture
            This updated Gaussian mixture.

        Raises
        ------
        ValueError
            If this mixture is not two-dimensional or a rotation parameter is
            invalid.
        """
        expected_dim = 2
        if self.dim != expected_dim:
            msg = "rotate_2d_ requires a two-dimensional GaussianMixture"
            raise ValueError(msg)
        if isinstance(angle, Tensor) and angle.device != self.device:
            msg = "angle and GaussianMixture must be on the same device"
            raise ValueError(msg)

        rotation_dtype = _promoted_dtype(self.means, self.covariances)
        if isinstance(angle, Tensor) and not angle.is_complex():
            rotation_dtype = torch.promote_types(rotation_dtype, angle.dtype)
        if center is not None and not center.is_complex():
            rotation_dtype = torch.promote_types(rotation_dtype, center.dtype)
        matrix = rotation_matrix_2d(angle, dtype=rotation_dtype, device=self.device)
        offset = self._rotation_offset(matrix, center)
        return self.affine_transform_(matrix, offset)

    def rotate_3d_(
        self, axis: Tensor, angle: float | Tensor, center: Tensor | None = None
    ) -> "GaussianMixture":
        """Rotate this three-dimensional Gaussian mixture in place.

        Parameters
        ----------
        axis : Tensor
            Finite, non-zero real vector with shape ``(3,)``. It is normalized
            internally.
        angle : float or Tensor
            Finite real scalar angle in radians following the right-hand rule.
        center : Tensor, optional
            Rotation center with shape ``(3,)`` on the same device as this
            mixture. The origin is used by default.

        Returns
        -------
        GaussianMixture
            This updated Gaussian mixture.

        Raises
        ------
        ValueError
            If this mixture is not three-dimensional or a rotation parameter
            is invalid.
        """
        expected_dim = 3
        if self.dim != expected_dim:
            msg = "rotate_3d_ requires a three-dimensional GaussianMixture"
            raise ValueError(msg)
        if axis.device != self.device:
            msg = "axis and GaussianMixture must be on the same device"
            raise ValueError(msg)
        if isinstance(angle, Tensor) and angle.device != self.device:
            msg = "angle and GaussianMixture must be on the same device"
            raise ValueError(msg)

        rotation_dtype = _promoted_dtype(self.means, self.covariances)
        for tensor in (axis, angle) if isinstance(angle, Tensor) else (axis,):
            if not tensor.is_complex():
                rotation_dtype = torch.promote_types(rotation_dtype, tensor.dtype)
        if center is not None and not center.is_complex():
            rotation_dtype = torch.promote_types(rotation_dtype, center.dtype)
        matrix = rotation_matrix_3d(axis, angle, dtype=rotation_dtype, device=self.device)
        offset = self._rotation_offset(matrix, center)
        return self.affine_transform_(matrix, offset)



    def _rotation_offset(self, matrix: Tensor, center: Tensor | None) -> Tensor:
        """Return the affine offset for rotation around ``center``."""
        if center is None:
            return torch.zeros(self.dim, dtype=matrix.dtype, device=self.device)
        if center.ndim != 1 or center.shape[0] != self.dim:
            msg = "center must have shape (dim)"
            raise ValueError(msg)
        if center.device != self.device:
            msg = "center and GaussianMixture must be on the same device"
            raise ValueError(msg)
        if center.is_complex():
            msg = "center must be real"
            raise ValueError(msg)
        if not torch.isfinite(center).all():
            msg = "center must contain only finite values"
            raise ValueError(msg)

        dtype = torch.promote_types(matrix.dtype, center.dtype)
        matrix = matrix.to(dtype=dtype)
        center = center.to(dtype=dtype)
        return center - matrix @ center

    def affine_transform_(self, matrix: Tensor, offset: Tensor) -> "GaussianMixture":
        """Apply an affine transformation to this Gaussian mixture in place.

        The transformation updates this mixture in place by mapping each random vector ``x`` to
        ``matrix @ x + offset``. Component weights are preserved, while means
        and covariances are transformed in batched form.

        Parameters
        ----------
        matrix : Tensor
            Transformation matrix with shape ``(output_dim, self.dim)``.
        offset : Tensor
            Translation vector with shape ``(output_dim,)``.

        Returns
        -------
        GaussianMixture
            This updated Gaussian mixture.

        Raises
        ------
        ValueError
            If the parameters have incompatible shapes or devices, or if the matrix
            does not have full row rank.
        """
        if torch.linalg.matrix_rank(matrix.to(dtype=self.means.dtype)) < matrix.shape[0]:
            msg = "matrix must have full row rank"
            raise ValueError(msg)

        means, covariances = _affine_transform_parameters(
            matrix, offset, self.means, self.covariances
        )

        self._replace_parameters(
            self.n_components, matrix.shape[0], self.weights, means, covariances
        )
        return self

    def add_independent_(self, gm1: "GaussianMixture") -> "GaussianMixture":
        """Update this mixture in place to the sum with an independent Gaussian mixture.

        Parameters
        ----------
        gm1 : GaussianMixture
            Independent Gaussian mixture to add. It must have the same dimension
            and device as this mixture.

        Returns
        -------
        GaussianMixture
            This updated Gaussian mixture.

        Raises
        ------
        ValueError
            If the input dimensions or devices differ.

        Notes
        -----
        Independence is a semantic requirement and cannot be inferred from the
        marginal distributions. The result has
        ``self.n_components * gm1.n_components`` components.
        """
        if self.dim != gm1.dim:
            msg = "GaussianMixture dimensions must match for independent addition"
            raise ValueError(msg)
        if self.device != gm1.device:
            msg = "GaussianMixture devices must match for independent addition"
            raise ValueError(msg)

        weights, means, covariances = _sum_independent_parameters(
            self.weights, self.means, self.covariances, gm1.weights, gm1.means, gm1.covariances
        )

        self._replace_parameters(
            self.n_components * gm1.n_components, self.dim, weights, means, covariances
        )
        return self

    def cond(self, obs_values: Tensor, obs_idx: Tensor | None = None) -> "GaussianMixture":
        """Condition a Gaussian mixture on y observed dimensions.

        If `obs_idx` is None, assumes the observed dimensions are the last y
        dimensions, where y = obs_values.shape[-1].
        """
        obs_values = obs_values.to(self.device)

        k = self.n_components
        n_dims = self.dim

        if n_dims <= 1:
            msg = "conditioning is only possible on a gmm of dimension 2 or higher"
            raise ValueError(msg)

        obs_values = self._validate_obs_values(obs_values, n_dims)
        n_obs = obs_values.shape[0]
        n_unobs = n_dims - n_obs

        all_means = self.means
        all_covs = self.covariances
        all_weights = self.weights

        if obs_idx is not None:
            obs_idx = self._validate_obs_idx(obs_idx, obs_values, n_dims)
            obs_idx = obs_idx.to(self.device)

            # calculates permutation vector
            mask = torch.ones(n_dims, dtype=torch.bool, device=self.device)
            mask[obs_idx] = False
            unobs_idx = torch.nonzero(mask, as_tuple=False).squeeze(1)
            perm = torch.cat((unobs_idx, obs_idx))

            all_means = all_means[:, perm]
            all_covs = all_covs[:, perm, :][:, :, perm]  # permute rows then permute columns

        means_x = all_means[:, :n_unobs]
        means_y = all_means[:, n_unobs:]
        covs_xx = all_covs[:, :n_unobs, :n_unobs]
        covs_xy = all_covs[:, :n_unobs, n_unobs:]
        covs_yx = covs_xy.transpose(1, 2)
        covs_yy = all_covs[:, n_unobs:, n_unobs:]

        mean_diff = obs_values.unsqueeze(0) - means_y

        try:
            cholesky = torch.linalg.cholesky(covs_yy)
            # calculates cov_yy^-1 * (obs_values - means_y)
            yy_inv_diff = torch.cholesky_solve(mean_diff.unsqueeze(-1), cholesky)
            # calculates cov_yy^-1 * cov_yx
            yy_inv_yx = torch.cholesky_solve(covs_yx, cholesky)
            # constructs the marginal distribution needed for weights
            dist = torch.distributions.MultivariateNormal(loc=means_y, scale_tril=cholesky)
        except torch.linalg.LinAlgError as err:  # type: ignore[attr-defined]
            """
            The full covariance matrix is previosly validated to be positive definite
            (in the constructor), therefore this error is not possible mathematically
            speaking (as cov_yy is a principal submatrix of the full covariance matrix),
            yet if the full covariance matrix is close to singular, numerical issues may
            arise now. In this case, we raise an error to alert the user that the conditioning
            operation cannot be performed due to numerical instability.
            """
            min_eigval = torch.linalg.eigvalsh(covs_yy).min()
            min_eigval_full = torch.linalg.eigvalsh(all_covs).min()
            msg = (
                f"Cholesky decomposition failed during conditioning. This may be due to numerical instability."
                f"The smallest eigenvalue of the cov_yy matrix is {min_eigval.item()}. "
                f"The smallest eigenvalue of the full covariance matrix is {min_eigval_full.item()}."
            )
            raise RuntimeError(msg) from err

        means_cond = means_x + (covs_xy @ yy_inv_diff).squeeze(-1)
        covs_cond = covs_xx - covs_xy @ yy_inv_yx

        logprob = dist.log_prob(obs_values)  # type: ignore[no-untyped-call]
        log_weights_cond = torch.log(all_weights) + logprob
        weights_cond = torch.softmax(log_weights_cond, dim=0)

        return GaussianMixture(k, n_unobs, weights_cond, means_cond, covs_cond, device=self.device)

    def marginalize(self, idx_keep: Tensor) -> "GaussianMixture":
        """Marginalize a Gaussian mixture to the coordinates in ``idx_keep``."""
        idx_keep = self._validate_idx(idx_keep, self.dim).to(self.device)

        means_marg = self.means[:, idx_keep]
        covs_marg = self.covariances[:, idx_keep, :][:, :, idx_keep]

        return GaussianMixture(
            n_components=self.n_components,
            dim=idx_keep.shape[0],
            weights=self.weights,
            means=means_marg,
            covariances=covs_marg,
            device=self.device,
        )



    def _validate_idx(self, obs_idx: Tensor, n_dims: int) -> Tensor:
        if obs_idx.ndim != 1:
            msg = "observed indices must be a vector"
            raise ValueError(msg)
        if obs_idx.shape[0] >= n_dims:
            msg = "number of observed indices must be smaller than total number of dimensions"
            raise ValueError(msg)
        if obs_idx.dtype not in (torch.int8, torch.int16, torch.int32, torch.int64):
            msg = "observed indices must be integers"
            raise ValueError(msg)
        if (obs_idx.max() > n_dims - 1) or (obs_idx.min() < 0):
            msg = "out of bound value for observed index"
            raise ValueError(msg)
        has_duplicates = obs_idx.unique().numel() != obs_idx.numel()  # type: ignore[no-untyped-call]
        if has_duplicates:
            msg = "each observed index must be unique"
            raise ValueError(msg)
        return obs_idx

    def _validate_obs_values(self, obs_values: Tensor, n_dim: int) -> Tensor:
        if obs_values.ndim != 1:
            msg = "observed values must be a vector"
            raise ValueError(msg)
        if obs_values.shape[0] >= n_dim:
            msg = "number of observed values must be smaller than total number of dimensions"
            raise ValueError(msg)
        if obs_values.device != self.means.device:
            msg = f"observed values are on {obs_values.device} but GMM is on {self.means.device}"
            raise ValueError(msg)
        if not obs_values.is_floating_point():
            msg = "observed values must be a floating point tensor"
            raise ValueError(msg)
        return obs_values

    def _validate_obs_idx(self, obs_idx: Tensor, obs_values: Tensor, n_dims: int) -> Tensor:
        if obs_idx.shape[0] != obs_values.shape[0]:
            msg = "number of observed indices must correspond to the number of observed values"
            raise ValueError(msg)
        return self._validate_idx(obs_idx, n_dims)

    def _validate_n_components(self, k: int) -> int:
        if not isinstance(k, int) or k <= 0:
            msg = "n_components must be a positive integer"
            raise ValueError(msg)
        return k

    def _validate_dim(self, d: int) -> int:
        if not isinstance(d, int) or d <= 0:
            msg = "dim must be a positive integer"
            raise ValueError(msg)
        return d

    def _validate_weights(self, w: Tensor, n_components: int) -> Tensor:
        if w.ndim != 1 or w.shape[0] != n_components:
            msg = "weights must have shape (n_components)"
            raise ValueError(msg)
        w = w.to(self.device)
        w_sum = w.sum()
        if not torch.isclose(w_sum, torch.ones((), device=self.device, dtype=w.dtype)):
            msg = "weights must sum to 1.0"
            raise ValueError(msg)
        if torch.any(w < 0):
            msg = "weights must be non-negative"
            raise ValueError(msg)
        if w_sum <= 0:
            msg = "weights must sum to a positive value"
            raise ValueError(msg)
        return (w / w_sum).clone()

    def _validate_means(self, m: Tensor, n_components: int, dim: int) -> Tensor:
        mean_tensor_order = 2
        if m.ndim != mean_tensor_order or m.shape != (n_components, dim):
            msg = "means must have shape (n_components, dim)"
            raise ValueError(msg)
        return m.to(self.device).clone()

    def _validate_covariances(self, c: Tensor, n_components: int, dim: int) -> Tensor:
        covariance_tensor_order = 3
        if c.ndim != covariance_tensor_order or c.shape != (n_components, dim, dim):
            msg = "covariances must have shape (n_components, dim, dim)"
            raise ValueError(msg)
        c = c.to(self.device).clone()
        # Symmetry check
        if not torch.allclose(c, c.transpose(1, 2)):
            msg = "Covariance matrices must be symmetric"
            raise ValueError(msg)
        # Basic positive-definite check (diagonal > 0 if diagonal matrix)
        for i in range(n_components):
            matrix = c[i, :, :]
            eigenvalues = torch.linalg.eigvalsh(matrix)
            if torch.any(eigenvalues <= 0):
                msg = "Covariance matrices must be positive definite"
                raise ValueError(msg)
        return c

            f"means_shape={tuple(self.means.shape)}, "
            f"covariances_shape={tuple(self.covariances.shape)}, device={self.device})"
        )
