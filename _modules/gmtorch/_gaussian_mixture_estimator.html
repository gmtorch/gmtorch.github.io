"""PyTorch EM estimator skeleton for Gaussian mixture models."""

from __future__ import annotations

import math
from dataclasses import dataclass
from typing import cast

import torch
from torch import Tensor

from gmtorch._em import BaseMixtureEM, InitMethod
from gmtorch._gaussian_mixture import GaussianMixture

MATRIX_TENSOR_ORDER = 2
COVARIANCE_TENSOR_ORDER = 3
COMPONENT_COUNT_EPS_FACTOR = 10


@dataclass(slots=True)
class GaussianMixtureEMState:
    """Learned full-covariance Gaussian mixture parameters.

    Attributes
    ----------
    weights : Tensor
        Normalized mixture weights of shape ``(n_components,)``.
    means : Tensor
        Component means of shape ``(n_components, dim)``.
    covariances : Tensor
        Full component covariance matrices of shape ``(n_components, dim, dim)``.
    precision_cholesky : Tensor
        Cholesky factors used for stable precision-based log-probability evaluation.

    Notes
    -----
    All tensors should live on the estimator device and use the estimator dtype. The
    covariance representation intentionally matches ``GaussianMixture`` so fitted state
    can be converted to the existing distribution object without changing its API.
    """

    weights: Tensor
    means: Tensor
    covariances: Tensor
    precision_cholesky: Tensor

class GaussianMixtureEstimator(BaseMixtureEM):
    """PyTorch EM estimator for full-covariance Gaussian mixtures.

    This class should own fitting state and training metadata. The existing
    ``GaussianMixture`` class should remain the distribution/value object, while this
    estimator learns parameters and exposes ``to_mixture`` to build that distribution.

    Conceptual responsibilities:

    - parameter initialization through ``_initialize_parameters`` and
      ``_initial_responsibilities``;
    - E-step orchestration inherited from ``BaseMixtureEM``;
    - M-step parameter updates through ``_m_step``;
    - log-probability computation through ``_estimate_log_component_prob``;
    - covariance estimation through module-level covariance helpers;
    - prediction and responsibility computation inherited from ``BaseMixtureEM``;
    - model state storage through ``GaussianMixtureEMState``.

    Examples
    --------
    >>> import torch
    >>> from gmtorch import GaussianMixtureEstimator
    >>> x = torch.tensor([[-2.0], [-1.8], [-2.2], [2.0], [1.9], [2.1]])
    >>> estimator = GaussianMixtureEstimator(n_components=2, max_iter=3, seed=0)
    >>> labels = estimator.fit_predict(x)
    >>> assert labels.shape == (6,)
    >>> mixture = estimator.to_mixture()
    >>> assert mixture.n_components == 2
    """

    def __init__(  # noqa: PLR0913
        self,
        n_components: int = 1,
        *,
        reg_covar: float = 1e-6,
        tol: float = 1e-3,
        max_iter: int = 100,
        n_init: int = 1,
        init_method: InitMethod = "random_from_data",
        weights_init: Tensor | None = None,
        means_init: Tensor | None = None,
        covariances_init: Tensor | None = None,
        seed: int | None = None,
        warm_start: bool = False,
        device: torch.device | str | None = None,
        dtype: torch.dtype | None = None,
    ) -> None:
        """Store Gaussian EM configuration and optional initial parameters."""
        if reg_covar < 0:
            msg = "reg_covar must be non-negative"
            raise ValueError(msg)

        super().__init__(
            n_components=n_components,
            tol=tol,
            max_iter=max_iter,
            n_init=n_init,
            init_method=init_method,
            seed=seed,
            warm_start=warm_start,
            device=device,
            dtype=dtype,
        )
        self.reg_covar = float(reg_covar)
        self.weights_init = weights_init
        self.means_init = means_init
        self.covariances_init = covariances_init
        self._state: GaussianMixtureEMState | None = None
        self.dim = None

    @property
    def state(self) -> GaussianMixtureEMState:
        """Return fitted model state or raise if the estimator has not been fit.

        Examples
        --------
        >>> import torch
        >>> from gmtorch import GaussianMixtureEstimator
        >>> x = torch.tensor([[-1.0], [-0.8], [0.9], [1.1]])
        >>> estimator = GaussianMixtureEstimator(n_components=2, max_iter=2, seed=0)
        >>> _ = estimator.fit(x)
        >>> assert estimator.state.means.shape == (2, 1)
        """
        if self._state is None:
            msg = "GaussianMixtureEstimator has not been fit"
            raise ValueError(msg)
        return self._state

    def to_mixture(self) -> GaussianMixture:
        """Build a ``GaussianMixture`` distribution from the fitted EM state.

        Examples
        --------
        >>> import torch
        >>> from gmtorch import GaussianMixtureEstimator
        >>> x = torch.tensor([[-1.0], [-0.8], [0.9], [1.1]])
        >>> estimator = GaussianMixtureEstimator(n_components=2, max_iter=2, seed=0)
        >>> _ = estimator.fit(x)
        >>> gm = estimator.to_mixture()
        >>> assert gm.means.shape == (2, 1)
        """
        state = self.state
        return GaussianMixture(
            n_components=state.weights.shape[0],
            dim=state.means.shape[1],
            weights=state.weights,
            means=state.means,
            covariances=state.covariances,
            device=state.weights.device,
        )



    def _validate_initial_parameters(self, x: Tensor) -> None:
        """Validate optional initial weights, means, and covariances.

        Validation should mirror the public ``GaussianMixture`` constraints where
        possible: normalized non-negative weights, means shaped ``(n_components, dim)``,
        and symmetric positive-definite full covariance matrices.
        """
        x = self._prepare_input(x)
        dim = x.shape[1]

        if self.weights_init is not None:
            self.weights_init = _validate_initial_weights(
                self.weights_init, self.n_components, device=x.device, dtype=x.dtype
            )

        if self.means_init is not None:
            self.means_init = _validate_initial_means(
                self.means_init, self.n_components, dim, device=x.device, dtype=x.dtype
            )

        if self.covariances_init is not None:
            self.covariances_init = _validate_initial_covariances(
                self.covariances_init, self.n_components, dim, device=x.device, dtype=x.dtype
            )

    def _initialize_parameters(self, x: Tensor, generator: torch.Generator | None) -> None:
        """Initialize state from user parameters or generated responsibilities."""
        x = self._prepare_input(x)
        self._validate_initial_parameters(x)

        weights_init = self.weights_init
        means_init = self.means_init
        covariances_init = self.covariances_init

        if weights_init is None or means_init is None or covariances_init is None:
            responsibilities = self._initial_responsibilities(x, generator)
            self._initialize_from_responsibilities(x, responsibilities)
            state = self.state
            weights = state.weights if weights_init is None else weights_init
            means = state.means if means_init is None else means_init
            covariances = state.covariances if covariances_init is None else covariances_init
        else:
            weights = weights_init
            means = means_init
            covariances = covariances_init

        self._state = GaussianMixtureEMState(
            weights=weights.clone(),
            means=means.clone(),
            covariances=covariances.clone(),
            precision_cholesky=_compute_precision_cholesky(covariances),
        )

    def _initial_responsibilities(self, x: Tensor, generator: torch.Generator | None) -> Tensor:
        """Create initial responsibilities according to ``init_method``.

        Supported methods should be PyTorch-native. A future implementation may use
        random responsibilities, sampled data points, or k-means++ style seeding while
        keeping tensors on the configured device and dtype.
        """
        x = self._prepare_input(x)
        n_samples = x.shape[0]

        if self.init_method == "random":
            responsibilities = torch.rand(
                (n_samples, self.n_components), device=x.device, dtype=x.dtype, generator=generator
            )
            return responsibilities / responsibilities.sum(dim=1, keepdim=True)

        if self.init_method == "random_from_data":
            responsibilities = torch.zeros(
                (n_samples, self.n_components), device=x.device, dtype=x.dtype
            )
            indices = torch.randperm(n_samples, device=x.device, generator=generator)[
                : self.n_components
            ]
            components = torch.arange(self.n_components, device=x.device)
            responsibilities[indices, components] = 1
            return responsibilities

        msg = "kmeans++ initialization is not implemented yet"
        raise NotImplementedError(msg)

    def _initialize_from_responsibilities(self, x: Tensor, responsibilities: Tensor) -> None:
        """Estimate initial state from responsibilities produced during initialization."""
        x = self._prepare_input(x)
        expected_shape = (x.shape[0], self.n_components)
        if responsibilities.ndim != MATRIX_TENSOR_ORDER or responsibilities.shape != expected_shape:
            msg = "responsibilities must have shape (n_samples, n_components)"
            raise ValueError(msg)

        self._state = _estimate_gaussian_parameters(x, responsibilities, reg_covar=self.reg_covar)

    def _m_step(self, x: Tensor, log_resp: Tensor) -> None:
        """Run the Gaussian M-step and replace the current EM state."""
        x = self._prepare_input(x)
        expected_shape = (x.shape[0], self.n_components)
        if log_resp.ndim != MATRIX_TENSOR_ORDER or log_resp.shape != expected_shape:
            msg = "log_resp must have shape (n_samples, n_components)"
            raise ValueError(msg)
        if not log_resp.is_floating_point():
            msg = "log_resp must be a floating point tensor"
            raise ValueError(msg)

        self._state = _estimate_gaussian_parameters(x, log_resp.exp(), reg_covar=self.reg_covar)

    def _estimate_log_component_prob(self, x: Tensor) -> Tensor:
        """Compute Gaussian log probabilities using cached precision Cholesky factors."""
        return _estimate_log_full_gaussian_prob(x, self.state.means, self.state.precision_cholesky)

    def _estimate_log_weights(self) -> Tensor:
        """Return log normalized mixture weights from the current state."""
        return self.state.weights.log()




def _validate_initial_weights(
    weights: Tensor, n_components: int, *, device: torch.device, dtype: torch.dtype
) -> Tensor:
    """Validate and normalize user-supplied initial mixture weights."""
    if weights.ndim != 1 or weights.shape != (n_components,):
        msg = "weights_init must have shape (n_components,)"
        raise ValueError(msg)
    if not weights.is_floating_point():
        msg = "weights_init must be a floating point tensor"
        raise ValueError(msg)

    weights = weights.to(device=device, dtype=dtype)
    weights_sum = weights.sum()
    if torch.any(weights < 0):
        msg = "weights_init must be non-negative"
        raise ValueError(msg)
    if not torch.isclose(weights_sum, weights.new_tensor(1.0)):
        msg = "weights_init must sum to 1.0"
        raise ValueError(msg)

    return (weights / weights_sum).clone()


def _validate_initial_means(
    means: Tensor, n_components: int, dim: int, *, device: torch.device, dtype: torch.dtype
) -> Tensor:
    """Validate and align user-supplied initial Gaussian means."""
    if means.ndim != MATRIX_TENSOR_ORDER or means.shape != (n_components, dim):
        msg = "means_init must have shape (n_components, dim)"
        raise ValueError(msg)
    if not means.is_floating_point():
        msg = "means_init must be a floating point tensor"
        raise ValueError(msg)

    return means.to(device=device, dtype=dtype).clone()


def _validate_initial_covariances(
    covariances: Tensor, n_components: int, dim: int, *, device: torch.device, dtype: torch.dtype
) -> Tensor:
    """Validate and align user-supplied initial full covariance matrices."""
    if covariances.ndim != COVARIANCE_TENSOR_ORDER or covariances.shape != (n_components, dim, dim):
        msg = "covariances_init must have shape (n_components, dim, dim)"
        raise ValueError(msg)
    if not covariances.is_floating_point():
        msg = "covariances_init must be a floating point tensor"
        raise ValueError(msg)

    covariances = covariances.to(device=device, dtype=dtype)
    if not torch.allclose(covariances, covariances.transpose(-1, -2)):
        msg = "covariances_init must be symmetric"
        raise ValueError(msg)
    try:
        _compute_precision_cholesky(covariances)
    except ValueError as err:
        msg = "covariances_init must be positive definite"
        raise ValueError(msg) from err

    return covariances.clone()


def _estimate_gaussian_parameters(
    x: Tensor, responsibilities: Tensor, *, reg_covar: float
) -> GaussianMixtureEMState:
    """Estimate Gaussian mixture parameters for initialization or the M-step.

    Given samples and posterior responsibilities, this computes the
    full-covariance Gaussian maximum-likelihood updates used by EM. Component
    counts are stabilized with a small sklearn-style ``10 * eps`` offset before
    division, and the returned mixture weights are normalized to sum to one.

    Parameters
    ----------
    x : Tensor
        Input samples with shape ``(n_samples, dim)``. The tensor must be floating
        point, and the returned state preserves its device and dtype.
    responsibilities : Tensor
        Non-negative posterior component probabilities with shape
        ``(n_samples, n_components)``. Values are converted to the device and dtype
        of ``x`` before the estimates are computed.
    reg_covar : float
        Non-negative value added to every covariance diagonal for numerical
        regularization.

    Returns
    -------
    GaussianMixtureEMState
        Estimated full-covariance Gaussian mixture state. The state contains
        normalized ``weights`` with shape ``(n_components,)``, ``means`` with shape
        ``(n_components, dim)``, full ``covariances`` with shape
        ``(n_components, dim, dim)``, and matching ``precision_cholesky`` factors.

    Raises
    ------
    ValueError
        If inputs have incompatible shapes, are not floating point, contain negative
        responsibilities, use a negative regularization value, or produce covariance
        matrices that cannot be Cholesky-factorized.
    """
    if x.ndim != MATRIX_TENSOR_ORDER:
        msg = "x must have shape (n_samples, dim)"
        raise ValueError(msg)
    if responsibilities.ndim != MATRIX_TENSOR_ORDER:
        msg = "responsibilities must have shape (n_samples, n_components)"
        raise ValueError(msg)
    if not x.is_floating_point():
        msg = "x must be a floating point tensor"
        raise ValueError(msg)
    if not responsibilities.is_floating_point():
        msg = "responsibilities must be a floating point tensor"
        raise ValueError(msg)
    if reg_covar < 0:
        msg = "reg_covar must be non-negative"
        raise ValueError(msg)

    n_samples = x.shape[0]
    if responsibilities.shape[0] != n_samples:
        msg = "responsibilities must have shape (n_samples, n_components)"
        raise ValueError(msg)

    responsibilities = responsibilities.to(device=x.device, dtype=x.dtype)
    if torch.any(responsibilities < 0):
        msg = "responsibilities must be non-negative"
        raise ValueError(msg)

    component_counts = responsibilities.sum(dim=0)
    component_counts = component_counts + (
        COMPONENT_COUNT_EPS_FACTOR * torch.finfo(responsibilities.dtype).eps
    )

    weights = component_counts / component_counts.sum()
    means = (responsibilities.T @ x) / component_counts.unsqueeze(1)
    covariances = _estimate_full_covariances(
        x, responsibilities, component_counts, means, reg_covar=reg_covar
    )
    precision_cholesky = _compute_precision_cholesky(covariances)

    return GaussianMixtureEMState(
        weights=weights, means=means, covariances=covariances, precision_cholesky=precision_cholesky
    )


def _estimate_full_covariances(  # noqa: C901
    x: Tensor,
    responsibilities: Tensor,
    component_counts: Tensor,
    means: Tensor,
    *,
    reg_covar: float,
) -> Tensor:
    """Estimate full covariance matrices with diagonal regularization.

    This uses the responsibility-weighted maximum-likelihood covariance estimator
    from the Gaussian-mixture EM M-step:

    ``Sigma_k = sum_i r_ik (x_i - mu_k) (x_i - mu_k)^T / N_k``

    where ``r_ik`` is the responsibility of component ``k`` for sample ``i`` and
    ``N_k = sum_i r_ik`` is the effective component count. After the weighted
    estimate is computed, ``reg_covar`` is added to the diagonal to keep the
    empirical covariance away from singularity.

    This function should preserve the input device and dtype and return covariance
    matrices shaped ``(n_components, dim, dim)``.

    Parameters
    ----------
    x : Tensor
        Input samples of shape ``(n_samples, dim)``.
    responsibilities : Tensor
        Posterior component responsibilities of shape
        ``(n_samples, n_components)``.
    component_counts : Tensor
        Effective component counts of shape ``(n_components,)``.
    means : Tensor
        Component means of shape ``(n_components, dim)``.
    reg_covar : float
        Non-negative regularization added to each covariance diagonal.

    Returns
    -------
    Tensor
        Full covariance matrices of shape ``(n_components, dim, dim)``.
    """
    if x.ndim != MATRIX_TENSOR_ORDER:
        msg = "x must have shape (n_samples, dim)"
        raise ValueError(msg)
    if responsibilities.ndim != MATRIX_TENSOR_ORDER:
        msg = "responsibilities must have shape (n_samples, n_components)"
        raise ValueError(msg)
    if means.ndim != MATRIX_TENSOR_ORDER:
        msg = "means must have shape (n_components, dim)"
        raise ValueError(msg)
    if component_counts.ndim != 1:
        msg = "component_counts must have shape (n_components,)"
        raise ValueError(msg)
    if reg_covar < 0:
        msg = "reg_covar must be non-negative"
        raise ValueError(msg)

    n_samples, dim = x.shape
    n_components = means.shape[0]

    if responsibilities.shape != (n_samples, n_components):
        msg = "responsibilities must have shape (n_samples, n_components)"
        raise ValueError(msg)
    if means.shape[1] != dim:
        msg = "means must have shape (n_components, dim)"
        raise ValueError(msg)
    if component_counts.shape[0] != n_components:
        msg = "component_counts must have shape (n_components,)"
        raise ValueError(msg)
    if torch.any(component_counts <= 0):
        msg = "component_counts must be strictly positive"
        raise ValueError(msg)

    responsibilities = responsibilities.to(device=x.device, dtype=x.dtype)
    component_counts = component_counts.to(device=x.device, dtype=x.dtype)
    means = means.to(device=x.device, dtype=x.dtype)

    covariances = torch.empty((n_components, dim, dim), device=x.device, dtype=x.dtype)
    for component in range(n_components):
        diff = x - means[component]
        weighted_diff = responsibilities[:, component].unsqueeze(0) * diff.T
        covariances[component] = weighted_diff @ diff / component_counts[component]
        covariances[component].diagonal().add_(reg_covar)

    return covariances


def _compute_precision_cholesky(covariances: Tensor) -> Tensor:
    """Compute Cholesky factors of full covariance precision matrices.

    For each covariance matrix ``Sigma_k``, this computes a factor ``U_k`` such that
    ``U_k @ U_k.T = inv(Sigma_k)``. The factor is obtained with the same convention used
    by the Gaussian-mixture log-probability helper: if ``L_k = cholesky(Sigma_k)`` is the
    lower-triangular covariance Cholesky factor, then ``U_k = solve(L_k, I).T``.

    Parameters
    ----------
    covariances : Tensor
        Full covariance matrices of shape ``(n_components, dim, dim)``. The tensor must
        be floating point and each matrix must be symmetric positive definite.

    Returns
    -------
    Tensor
        Precision Cholesky factors of shape ``(n_components, dim, dim)`` on the same
        device and with the same dtype as ``covariances``.

    Raises
    ------
    ValueError
        If ``covariances`` does not have shape ``(n_components, dim, dim)``, is not
        floating point, is not square/symmetric, or contains a matrix that cannot be
        Cholesky-factorized.
    """
    if covariances.ndim != COVARIANCE_TENSOR_ORDER:
        msg = "covariances must have shape (n_components, dim, dim)"
        raise ValueError(msg)
    if covariances.shape[1] != covariances.shape[2]:
        msg = "covariances must have shape (n_components, dim, dim)"
        raise ValueError(msg)
    if not covariances.is_floating_point():
        msg = "covariances must be floating point tensors"
        raise ValueError(msg)
    if not torch.allclose(covariances, covariances.transpose(-1, -2)):
        msg = "covariances must be symmetric"
        raise ValueError(msg)

    n_components, dim, _ = covariances.shape
    estimate_precision_msg = (
        "Fitting the mixture model failed because some components have ill-defined "
        "empirical covariance. Try decreasing n_components, increasing reg_covar, "
        "or scaling the input data."
    )
    if covariances.dtype == torch.float32:
        estimate_precision_msg = (
            f"{estimate_precision_msg} Numerical accuracy can also be improved by using "
            "float64 data instead of float32."
        )

    try:
        covariance_cholesky = torch.linalg.cholesky(covariances)
    except torch.linalg.LinAlgError as err:  # type: ignore[attr-defined]
        raise ValueError(estimate_precision_msg) from err

    identity = torch.eye(dim, device=covariances.device, dtype=covariances.dtype)
    inverse_cholesky = torch.linalg.solve_triangular(
        covariance_cholesky, identity.expand(n_components, dim, dim), upper=False
    )
    return cast("Tensor", inverse_cholesky.transpose(-1, -2))


def _estimate_log_full_gaussian_prob(
    x: Tensor, means: Tensor, precision_cholesky: Tensor
) -> Tensor:
    """Return per-component full-covariance Gaussian log probabilities.

    This evaluates the Gaussian component log density for every sample and component:

    ``log N(x_i | mu_k, Sigma_k)``

    The computation uses the precision Cholesky factors returned by
    ``_compute_precision_cholesky``. For each component, the squared Mahalanobis term is
    computed as ``||(x_i - mu_k) @ U_k||^2``, where
    ``U_k @ U_k.T = inv(Sigma_k)``. The log determinant contribution is computed from the
    diagonal of ``U_k``.

    Parameters
    ----------
    x : Tensor
        Input samples of shape ``(n_samples, dim)``.
    means : Tensor
        Component means of shape ``(n_components, dim)``.
    precision_cholesky : Tensor
        Precision Cholesky factors of shape ``(n_components, dim, dim)``.

    Returns
    -------
    Tensor
        Component log probabilities of shape ``(n_samples, n_components)``.

    Raises
    ------
    ValueError
        If the input tensors do not have compatible full-covariance Gaussian shapes or
        are not floating point.
    """
    if x.ndim != MATRIX_TENSOR_ORDER:
        msg = "x must have shape (n_samples, dim)"
        raise ValueError(msg)
    if means.ndim != MATRIX_TENSOR_ORDER:
        msg = "means must have shape (n_components, dim)"
        raise ValueError(msg)
    if precision_cholesky.ndim != COVARIANCE_TENSOR_ORDER:
        msg = "precision_cholesky must have shape (n_components, dim, dim)"
        raise ValueError(msg)
    if not x.is_floating_point():
        msg = "x must be a floating point tensor"
        raise ValueError(msg)
    if not means.is_floating_point():
        msg = "means must be a floating point tensor"
        raise ValueError(msg)
    if not precision_cholesky.is_floating_point():
        msg = "precision_cholesky must be a floating point tensor"
        raise ValueError(msg)

    n_samples, dim = x.shape
    n_components = means.shape[0]

    if means.shape[1] != dim:
        msg = "means must have shape (n_components, dim)"
        raise ValueError(msg)
    if precision_cholesky.shape != (n_components, dim, dim):
        msg = "precision_cholesky must have shape (n_components, dim, dim)"
        raise ValueError(msg)

    means = means.to(device=x.device, dtype=x.dtype)
    precision_cholesky = precision_cholesky.to(device=x.device, dtype=x.dtype)

    log_det = torch.log(torch.diagonal(precision_cholesky, dim1=-2, dim2=-1)).sum(dim=1)
    squared_mahalanobis = torch.empty((n_samples, n_components), device=x.device, dtype=x.dtype)

    for component in range(n_components):
        transformed = (x @ precision_cholesky[component]) - (
            means[component] @ precision_cholesky[component]
        )
        squared_mahalanobis[:, component] = transformed.square().sum(dim=1)

    return -0.5 * (dim * math.log(2.0 * math.pi) + squared_mahalanobis) + log_det
