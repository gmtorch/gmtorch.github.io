"""
Divergence functions for Gaussian Mixtures.

This module implements several divergences used to compare
Gaussian Mixtures.
"""

import math

import torch
from torch import Tensor

from gmtorch._gaussian_mixture import GaussianMixture

def l2_divergence(p_gmm: GaussianMixture, q_gmm: GaussianMixture, block_size: int = 128) -> Tensor:
    """
    Compute the L2 divergence (a.k.a ISE, Integral Square Error) between two GMMs.

    Parameters
    ----------
    p_gmm : GaussianMixture
        First distribution.
    q_gmm : GaussianMixture
        Second distribution.
    block_size : int
        Block size used for Gaussian products. Larger values use more memory and
        can improve performance.

    Returns
    -------
    float
        The L2 divergence between p and q.
    """
    if p_gmm.dim != q_gmm.dim:
        msg = "Cannot compute L2 divergence for GMMs with different dimensions."
        raise ValueError(msg)

    j_p, j_pq, j_q = _compute_l2_quadratic_terms(p_gmm, q_gmm, block_size=block_size)
    ise = j_p - 2 * j_pq + j_q
    return ise.squeeze()

def cs_divergence(p_gmm: GaussianMixture, q_gmm: GaussianMixture, block_size: int = 128) -> Tensor:
    """
    Compute the Cauchy-Schwarz divergence between two GMMs.

    Parameters
    ----------
    p_gmm : GaussianMixture
        First distribution.
    q_gmm : GaussianMixture
        Second distribution.
    block_size : int
        Block size used for Gaussian products. Larger values use more memory and
        can improve performance.

    Returns
    -------
    float
        The Cauchy-Schwarz divergence between p and q.
    """
    if p_gmm.dim != q_gmm.dim:
        msg = "Cannot compute Cauchy-Schwarz divergence for GMMs with different dimensions."
        raise ValueError(msg)

    j_p, j_pq, j_q = _compute_l2_quadratic_terms(p_gmm, q_gmm, block_size=block_size)
    cs_div = -torch.log(j_pq**2 / j_p * j_q)
    return cs_div.squeeze()

def jr2_divergence(p_gmm: GaussianMixture, q_gmm: GaussianMixture, block_size: int = 128) -> Tensor:
    """
    Compute the Jeffrey-Renyi 2 divergence between two GMMs.

    Parameters
    ----------
    p_gmm : GaussianMixture
        First distribution.
    q_gmm : GaussianMixture
        Second distribution.
    block_size : int
        Block size used for Gaussian products. Larger values use more memory and
        can improve performance.

    Returns
    -------
    float
        The Jeffrey-Renyi 2 divergence between p and q.
    """
    if p_gmm.dim != q_gmm.dim:
        msg = "Cannot compute Jeffrey-Renyi 2 divergence for GMMs with different dimensions."
        raise ValueError(msg)

    j_p, j_pq, j_q = _compute_l2_quadratic_terms(p_gmm, q_gmm, block_size=block_size)
    jr2_div = -torch.log((j_p + 2 * j_pq + j_q) / 4) + 0.5 * (torch.log(j_p) + torch.log(j_q))
    return jr2_div.squeeze()

def tb2_divergence(p_gmm: GaussianMixture, q_gmm: GaussianMixture, block_size: int = 128) -> Tensor:
    """
    Compute the Total Bregman 2 divergence between two GMMs.

    Parameters
    ----------
    p_gmm : GaussianMixture
        First distribution.
    q_gmm : GaussianMixture
        Second distribution.
    block_size : int
        Block size used for Gaussian products. Larger values use more memory and
        can improve performance.

    Returns
    -------
    float
        The Total Bregman 2 divergence between p and q.
    """
    if p_gmm.dim != q_gmm.dim:
        msg = "Cannot compute Total Bregman 2 divergence for GMMs with different dimensions."
        raise ValueError(msg)

    j_p, j_pq, j_q = _compute_l2_quadratic_terms(p_gmm, q_gmm, block_size=block_size)
    tb2_div = (j_p - 2 * j_pq + j_q) / torch.sqrt(1 + 4 * j_q)
    return tb2_div.squeeze()




def _compute_l2_quadratic_terms(
    p_gmm: GaussianMixture, q_gmm: GaussianMixture, block_size: int = 128
) -> tuple[Tensor, Tensor, Tensor]:
    r"""
    Compute the quadratic terms needed for divergences like L2, Cauchy-Schwarz, Jeffrey-Renyi 2...

    Parameters
    ----------
    p_gmm : GaussianMixture
        First distribution.
    q_gmm : GaussianMixture
        Second distribution.
    block_size : int
        Block size used for Gaussian products. Larger values use more memory and
        can improve performance.

    Returns
    -------
    float
        The quadratic terms j_p = \int p(x)^2 dx, j_pq = \int p(x)q(x) dx, j_q = \int q(x)^2 dx
        needed for divergence computation

    """
    h_p = _calc_integral_self_gm_lower_blocked(p_gmm, block=block_size)
    h_pq = _calc_integral_outer_prod_gm_blocked(p_gmm, q_gmm, block=block_size)
    h_q = _calc_integral_self_gm_lower_blocked(q_gmm, block=block_size)

    j_p = p_gmm.weights @ h_p @ p_gmm.weights
    j_pq = p_gmm.weights @ h_pq @ q_gmm.weights
    j_q = q_gmm.weights @ h_q @ q_gmm.weights
    return j_p, j_pq, j_q


def _calc_integral_outer_prod_gm_blocked(
    gm0: GaussianMixture, gm1: GaussianMixture, block: int = 64, eps: float = 0.0
) -> Tensor:
    """
    Compute the M x N Gaussian product matrix.

    The matrix is defined as
        H[i, j] = exp(-0.5 * (d*log(2π) + log det(var_i + var_j)
                              + (mu_i - mu_j)^T (var_i + var_j)^(-1) (mu_i - mu_j)))

    without ever materializing the full (M, N, D, D) tensor.

    Returns
    -------
      H: (M, N)
    """
    assert gm0.dim == gm1.dim, "different dims"

    mu0 = gm0.means
    mu1 = gm1.means
    var0 = gm0.covariances
    var1 = gm1.covariances

    m, d = mu0.shape
    n = mu1.shape[0]

    device = mu0.device
    dtype = mu0.dtype

    h = torch.empty((m, n), device=device, dtype=dtype)

    if eps != 0.0:
        identity = torch.eye(d, device=device, dtype=dtype)

    const = d * math.log(2.0 * math.pi)

    for i0 in range(0, m, block):
        i1 = min(i0 + block, m)

        mu0_blk = mu0[i0:i1]  # (bm, D)
        var0_blk = var0[i0:i1]  # (bm, D, D)

        for j0 in range(0, n, block):
            j1 = min(j0 + block, n)

            mu1_blk = mu1[j0:j1]  # (bn, D)
            var1_blk = var1[j0:j1]  # (bn, D, D)

            # Pairwise differences: (bm, bn, D)
            diff = mu0_blk[:, None, :] - mu1_blk[None, :, :]

            # Pairwise covariance sums: (bm, bn, D, D)
            var_ij = var0_blk[:, None, :, :] + var1_blk[None, :, :, :]

            if eps != 0.0:
                var_ij = var_ij + eps * identity

            # Batched Cholesky over the whole block
            chol = torch.linalg.cholesky(var_ij)

            # Solve (var_ij) x = diff, without forming inverse
            # diff[..., None] is a tensor of shape (bm, bn, D, 1)
            sol = torch.cholesky_solve(diff[..., None], chol).squeeze(-1)  # (bm, bn, D)

            # Quadratic form: diff^T @ inv(var_ij) @ diff
            term1 = (diff * sol).sum(dim=-1)  # (bm, bn)

            # log det(var_ij) from Cholesky
            logdet = 2.0 * torch.log(torch.diagonal(chol, dim1=-2, dim2=-1)).sum(dim=-1)  # (bm, bn)

            h[i0:i1, j0:j1] = torch.exp(-0.5 * (const + logdet + term1))

    return h


def _calc_integral_self_gm_lower_blocked(  # noqa: C901, PLR0915
    gm: GaussianMixture, block: int = 128, eps: float = 0.0, *, fill_upper: bool = True
) -> Tensor:
    """
    Compute H for gm against itself, exploiting symmetry.

    Returns
    -------
        H: (M, M)
           - if fill_upper=False: lower triangle + diagonal filled, upper is 0
           - if fill_upper=True : full symmetric matrix is filled
    """
    mu = gm.means
    var = gm.covariances
    m, d = mu.shape

    device = mu.device
    dtype = mu.dtype

    h = torch.zeros((m, m), device=device, dtype=dtype)
    const = d * math.log(2.0 * math.pi)

    identity = None
    if eps != 0.0:
        identity = torch.eye(d, device=device, dtype=dtype)

    # ---- Diagonal entries H[i, i] ----
    # H_ii uses 2 * var_i and diff = 0, so no solve is needed.
    var_diag = 2.0 * var
    if eps != 0.0 and identity is not None:
        var_diag = var_diag + eps * identity

    chol_diag = torch.linalg.cholesky(var_diag)  # (M, D, D)
    logdet_diag = 2.0 * torch.log(torch.diagonal(chol_diag, dim1=-2, dim2=-1)).sum(dim=-1)  # (M,)

    h_diag = torch.exp(-0.5 * (const + logdet_diag))
    idx = torch.arange(m, device=device)
    h[idx, idx] = h_diag

    # ---- Strictly lower triangle, blockwise ----
    for i0 in range(0, m, block):
        i1 = min(i0 + block, m)

        mu_i = mu[i0:i1]
        var_i = var[i0:i1]

        for j0 in range(0, i0 + 1, block):
            j1 = min(j0 + block, m)

            # Off-diagonal block: full block is needed
            if j0 < i0:
                mu_j = mu[j0:j1]
                var_j = var[j0:j1]

                diff = mu_i[:, None, :] - mu_j[None, :, :]
                var_ij = var_i[:, None, :, :] + var_j[None, :, :, :]
                if eps != 0.0 and identity is not None:
                    var_ij = var_ij + eps * identity

                chol = torch.linalg.cholesky(var_ij)
                sol = torch.cholesky_solve(diff[..., None], chol).squeeze(-1)
                quad = (diff * sol).sum(dim=-1)
                logdet = 2.0 * torch.log(torch.diagonal(chol, dim1=-2, dim2=-1)).sum(dim=-1)
                block_vals = torch.exp(-0.5 * (const + logdet + quad))

                h[i0:i1, j0:j1] = block_vals
                if fill_upper:
                    h[j0:j1, i0:i1] = block_vals.transpose(0, 1)
                continue

            # Diagonal block: only strict lower triangle, since diagonal already done
            block_size = i1 - i0
            tril_idx = torch.tril_indices(block_size, block_size, offset=-1, device=mu_i.device)
            r, c = tril_idx[0], tril_idx[1]
            if r.numel() == 0:
                continue

            diff = mu_i[r] - mu_i[c]  # (T, D)
            var_rc = var_i[r] + var_i[c]  # (T, D, D)
            if eps != 0.0 and identity is not None:
                var_rc = var_rc + eps * identity

            chol = torch.linalg.cholesky(var_rc)  # (T, D, D)
            sol = torch.cholesky_solve(diff[..., None], chol).squeeze(-1)
            quad = (diff * sol).sum(dim=-1)  # (T,)
            logdet = 2.0 * torch.log(torch.diagonal(chol, dim1=-2, dim2=-1)).sum(dim=-1)
            vals = torch.exp(-0.5 * (const + logdet + quad))

            h[i0 + r, i0 + c] = vals
            if fill_upper:
                h[i0 + c, i0 + r] = vals

    return h
