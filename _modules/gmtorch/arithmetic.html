def rotation_matrix_2d(
    angle: float | Tensor,
    *,
    dtype: torch.dtype | None = None,
    device: torch.device | str | None = None,
) -> Tensor:
    """Construct a counterclockwise 2D rotation matrix.

    Parameters
    ----------
    angle : float or Tensor
        Finite real scalar angle in radians.
    dtype : torch.dtype, optional
        Floating-point dtype of the returned matrix. By default, a tensor
        angle's dtype is preserved and Python scalars use PyTorch's default
        floating-point dtype.
    device : torch.device or str, optional
        Device of the returned matrix. By default, a tensor angle's device is
        preserved and Python scalars use the CPU.

    Returns
    -------
    Tensor
        Rotation matrix with shape ``(2, 2)``.

    Examples
    --------
    >>> import torch
    >>> from gmtorch.arithmetic import rotation_matrix_2d
    >>> matrix = rotation_matrix_2d(torch.pi / 2)
    >>> rotated = matrix @ torch.tensor([1.0, 0.0])
    >>> assert torch.allclose(rotated, torch.tensor([0.0, 1.0]), atol=1e-6)
    """
    angle_tensor = _validate_rotation_angle(angle)
    rotation_dtype = _resolve_rotation_dtype(dtype, angle_tensor)
    angle_tensor = angle_tensor.to(dtype=rotation_dtype, device=device)

    cosine = torch.cos(angle_tensor)
    sine = torch.sin(angle_tensor)
    return torch.stack((torch.stack((cosine, -sine)), torch.stack((sine, cosine))))

def rotation_matrix_3d(
    axis: Tensor,
    angle: float | Tensor,
    *,
    dtype: torch.dtype | None = None,
    device: torch.device | str | None = None,
) -> Tensor:
    """Construct a 3D axis-angle rotation matrix using Rodrigues' formula.

    Positive angles follow the right-hand rule around ``axis``.

    Parameters
    ----------
    axis : Tensor
        Finite, non-zero real vector with shape ``(3,)``. It is normalized
        internally.
    angle : float or Tensor
        Finite real scalar angle in radians.
    dtype : torch.dtype, optional
        Floating-point dtype of the returned matrix. By default, the input
        tensor dtypes are promoted.
    device : torch.device or str, optional
        Device of the returned matrix. When omitted, ``axis.device`` is used
        and a tensor angle must already be on that device.

    Returns
    -------
    Tensor
        Rotation matrix with shape ``(3, 3)``.

    Examples
    --------
    >>> import torch
    >>> from gmtorch.arithmetic import rotation_matrix_3d
    >>> matrix = rotation_matrix_3d(torch.tensor([0.0, 0.0, 1.0]), torch.pi / 2)
    >>> rotated = matrix @ torch.tensor([1.0, 0.0, 0.0])
    >>> assert torch.allclose(rotated, torch.tensor([0.0, 1.0, 0.0]), atol=1e-6)
    """
    if axis.ndim != 1 or axis.shape[0] != 3:  # noqa: PLR2004
        msg = "axis must have shape (3)"
        raise ValueError(msg)
    if axis.is_complex():
        msg = "axis must be real"
        raise ValueError(msg)
    if not torch.isfinite(axis).all():
        msg = "axis must contain only finite values"
        raise ValueError(msg)

    angle_tensor = _validate_rotation_angle(angle)
    if device is None and isinstance(angle, Tensor) and angle_tensor.device != axis.device:
        msg = "axis and angle must be on the same device when device is not specified"
        raise ValueError(msg)

    dtype_inputs = (axis, angle_tensor) if isinstance(angle, Tensor) else (axis,)
    rotation_dtype = _resolve_rotation_dtype(dtype, *dtype_inputs)
    target_device = axis.device if device is None else torch.device(device)
    axis = axis.to(dtype=rotation_dtype, device=target_device)
    angle_tensor = angle_tensor.to(dtype=rotation_dtype, device=target_device)

    axis_norm = torch.linalg.vector_norm(axis)
    if not torch.isfinite(axis_norm):
        msg = "axis norm must be finite"
        raise ValueError(msg)
    if axis_norm == 0:
        msg = "axis must be non-zero"
        raise ValueError(msg)
    axis = axis / axis_norm

    axis_x, axis_y, axis_z = axis.unbind()
    zero = torch.zeros((), dtype=rotation_dtype, device=target_device)
    skew_symmetric = torch.stack(
        (
            torch.stack((zero, -axis_z, axis_y)),
            torch.stack((axis_z, zero, -axis_x)),
            torch.stack((-axis_y, axis_x, zero)),
        )
    )
    identity = torch.eye(3, dtype=rotation_dtype, device=target_device)
    sine = torch.sin(angle_tensor)
    cosine = torch.cos(angle_tensor)
    return identity + sine * skew_symmetric + (1 - cosine) * (skew_symmetric @ skew_symmetric)

def scale_gmm(gm: GaussianMixture, scale: float | Tensor) -> GaussianMixture:
    """Return a scaled copy of a Gaussian mixture.

    Parameters
    ----------
    gm : GaussianMixture
        Gaussian mixture to scale.
    scale : float or Tensor
        Finite, non-zero real scalar. A tensor scale must be zero-dimensional
        and on the same device as ``gm``.

    Returns
    -------
    GaussianMixture
        Gaussian mixture representing ``scale * x``. The input mixture is not
        modified.

    Raises
    ------
    ValueError
        If ``scale`` is not a finite, non-zero real scalar or is on a different
        device.

    Examples
    --------
    >>> import torch
    >>> from gmtorch import GaussianMixture
    >>> from gmtorch.arithmetic import scale_gmm
    >>> gm = GaussianMixture(dim=1, means=torch.tensor([[2.0]]))
    >>> scaled = scale_gmm(gm, 3.0)
    >>> assert torch.allclose(scaled.means, torch.tensor([[6.0]]))
    >>> assert torch.allclose(gm.means, torch.tensor([[2.0]]))
    """
    result = gm.clone()
    return result.scale_(scale)

def shift_gmm(gm: GaussianMixture, offset: Tensor) -> GaussianMixture:
    """Return a shifted copy of a Gaussian mixture.

    Parameters
    ----------
    gm : GaussianMixture
        Gaussian mixture to shift.
    offset : Tensor
        Finite real vector with shape ``(gm.dim,)`` on the same device as
        ``gm``.

    Returns
    -------
    GaussianMixture
        Gaussian mixture representing ``x + offset``. The input mixture is not
        modified.

    Raises
    ------
    ValueError
        If ``offset`` has an invalid shape, contains non-finite or complex
        values, or is on a different device.

    Examples
    --------
    >>> import torch
    >>> from gmtorch import GaussianMixture
    >>> from gmtorch.arithmetic import shift_gmm
    >>> gm = GaussianMixture(dim=2)
    >>> shifted = shift_gmm(gm, torch.tensor([1.0, -1.0]))
    >>> assert torch.allclose(shifted.means, torch.tensor([[1.0, -1.0]]))
    >>> assert torch.allclose(gm.means, torch.zeros(1, 2))
    """
    result = gm.clone()
    return result.shift_(offset)

def rotate_2d_gmm(
    gm: GaussianMixture, angle: float | Tensor, center: Tensor | None = None
) -> GaussianMixture:
    """Return a rotated copy of a two-dimensional Gaussian mixture.

    Parameters
    ----------
    gm : GaussianMixture
        Two-dimensional Gaussian mixture to rotate.
    angle : float or Tensor
        Finite real scalar angle in radians. Positive angles rotate
        counterclockwise.
    center : Tensor, optional
        Rotation center with shape ``(2,)``. The origin is used by default.

    Returns
    -------
    GaussianMixture
        Rotated Gaussian mixture. The input mixture is not modified.

    Examples
    --------
    >>> import torch
    >>> from gmtorch import GaussianMixture
    >>> from gmtorch.arithmetic import rotate_2d_gmm
    >>> gm = GaussianMixture(dim=2, means=torch.tensor([[1.0, 0.0]]))
    >>> rotated = rotate_2d_gmm(gm, torch.pi / 2)
    >>> assert torch.allclose(rotated.means, torch.tensor([[0.0, 1.0]]), atol=1e-6)
    >>> assert torch.allclose(gm.means, torch.tensor([[1.0, 0.0]]))
    """
    result = gm.clone()
    return result.rotate_2d_(angle, center)

def affine_transform_gmm(gm: GaussianMixture, matrix: Tensor, offset: Tensor) -> GaussianMixture:
    """Apply an affine transformation to a Gaussian mixture.

    Parameters
    ----------
    gm : GaussianMixture
        Gaussian mixture to transform.
    matrix : Tensor
        Transformation matrix with shape ``(output_dim, gm.dim)``.
    offset : Tensor
        Translation vector with shape ``(output_dim,)``.

    Returns
    -------
    GaussianMixture
        Gaussian mixture representing ``matrix @ x + offset``.
        The input mixture is not modified.

    Raises
    ------
    ValueError
        If the parameters have incompatible shapes or devices, or if the matrix
        does not have full row rank.

    Examples
    --------
    >>> import torch
    >>> from gmtorch import GaussianMixture
    >>> from gmtorch.arithmetic import affine_transform_gmm
    >>> gm = GaussianMixture(dim=2, means=torch.tensor([[1.0, 3.0]]))
    >>> transformed = affine_transform_gmm(gm, torch.tensor([[1.0, 0.0]]), torch.tensor([2.0]))
    >>> assert transformed.dim == 1
    >>> assert torch.allclose(transformed.means, torch.tensor([[3.0]]))
    >>> assert gm.dim == 2
    """
    result = gm.clone()
    return result.affine_transform_(matrix, offset)

def sum_independent_gmms(gm0: GaussianMixture, gm1: GaussianMixture) -> GaussianMixture:
    """Return the distribution of the sum of two independent Gaussian mixtures.

    Every pair of input components produces one output component. For component
    indices ``(i, j)``, the output weight is ``w0[i] * w1[j]``, the output mean
    is ``mu0[i] + mu1[j]``, and the output covariance is
    ``cov0[i] + cov1[j]``.

    Parameters
    ----------
    gm0 : GaussianMixture
        First Gaussian mixture.
    gm1 : GaussianMixture
        Second Gaussian mixture. It must have the same dimension and device as
        ``gm0``.

    Returns
    -------
    GaussianMixture
        Exact Gaussian-mixture distribution of the independent sum. Components
        are ordered with the ``gm1`` component index varying fastest. The input
        mixtures are not modified.

    Raises
    ------
    ValueError
        If the input dimensions or devices differ.

    Notes
    -----
    Independence is a semantic requirement and cannot be inferred from the two
    marginal distributions alone. The result has
    ``gm0.n_components * gm1.n_components`` components.

    Examples
    --------
    >>> import torch
    >>> from gmtorch import GaussianMixture
    >>> from gmtorch.arithmetic import sum_independent_gmms
    >>> gm0 = GaussianMixture(dim=1, means=torch.tensor([[1.0]]))
    >>> gm1 = GaussianMixture(dim=1, means=torch.tensor([[3.0]]))
    >>> summed = sum_independent_gmms(gm0, gm1)
    >>> assert torch.allclose(summed.means, torch.tensor([[4.0]]))
    >>> assert summed.n_components == gm0.n_components * gm1.n_components
    """
    result = gm0.clone()
    return result.add_independent_(gm1)

def substract_independent_gmms(gm0: GaussianMixture, gm1: GaussianMixture) -> GaussianMixture:
    """Return the distribution of the difference of two independent Gaussian mixtures.

    Every pair of input components produces one output component. For component
    indices ``(i, j)``, the output weight is ``w0[i] * w1[j]``, the output mean
    is ``mu0[i] - mu1[j]``, and the output covariance is
    ``cov0[i] + cov1[j]``.

    Parameters
    ----------
    gm0 : GaussianMixture
        Gaussian mixture from which ``gm1`` is subtracted.
    gm1 : GaussianMixture
        Independent Gaussian mixture to subtract. It must have the same
        dimension and device as ``gm0``.

    Returns
    -------
    GaussianMixture
        Exact Gaussian-mixture distribution of the independent difference. The
        input mixtures are not modified.

    Raises
    ------
    ValueError
        If the input dimensions or devices differ.

    Notes
    -----
    Independence is a semantic requirement and cannot be inferred from the two
    marginal distributions alone. The result has
    ``gm0.n_components * gm1.n_components`` components.

    Examples
    --------
    >>> import torch
    >>> from gmtorch import GaussianMixture
    >>> from gmtorch.arithmetic import substract_independent_gmms
    >>> gm0 = GaussianMixture(dim=1, means=torch.tensor([[1.0]]))
    >>> gm1 = GaussianMixture(dim=1, means=torch.tensor([[3.0]]))
    >>> difference = substract_independent_gmms(gm0, gm1)
    >>> assert torch.allclose(difference.means, torch.tensor([[-2.0]]))
    >>> assert difference.n_components == gm0.n_components * gm1.n_components
    """
    result = gm0.clone()
    return result.substract_independent_(gm1)




def _sum_independent_parameters(  # noqa: PLR0913
    weights0: Tensor,
    means0: Tensor,
    covariances0: Tensor,
    weights1: Tensor,
    means1: Tensor,
    covariances1: Tensor,
) -> tuple[Tensor, Tensor, Tensor]:
    """Compute the parameters of an independent Gaussian-mixture sum."""
    weight_dtype = torch.promote_types(weights0.dtype, weights1.dtype)
    component_dtype = _promoted_dtype(means0, covariances0, means1, covariances1)

    weights0 = weights0.to(dtype=weight_dtype)
    weights1 = weights1.to(dtype=weight_dtype)
    means0 = means0.to(dtype=component_dtype)
    means1 = means1.to(dtype=component_dtype)
    covariances0 = covariances0.to(dtype=component_dtype)
    covariances1 = covariances1.to(dtype=component_dtype)

    dim = means0.shape[1]
    weights = torch.outer(weights0, weights1).reshape(-1)
    means = (means0[:, None, :] + means1[None, :, :]).reshape(-1, dim)
    covariances = (covariances0[:, None, :, :] + covariances1[None, :, :, :]).reshape(-1, dim, dim)
    covariances = 0.5 * (covariances + covariances.transpose(-1, -2))

    return weights, means, covariances


def _affine_transform_parameters(
    matrix: Tensor, offset: Tensor, means: Tensor, covariances: Tensor
) -> tuple[Tensor, Tensor]:
    """Apply an affine transformation to batched Gaussian parameters.

    Parameters
    ----------
    matrix : Tensor
        Transformation matrix with shape ``(output_dim, input_dim)``.
    offset : Tensor
        Translation vector with shape ``(output_dim,)``.
    means : Tensor
        Batched means with shape ``(n_components, input_dim)``.
    covariances : Tensor
        Batched covariance matrices with shape
        ``(n_components, input_dim, input_dim)``.

    Returns
    -------
    tuple[Tensor, Tensor]
        Transformed means with shape ``(n_components, output_dim)`` and
        transformed covariances with shape
        ``(n_components, output_dim, output_dim)``.
    """
    matrix_order = 2
    vector_order = 1
    means_order = 2
    covariance_tensor_order = 3
    if matrix.ndim != matrix_order:
        msg = "matrix must have shape (output_dim, input_dim)"
        raise ValueError(msg)
    if offset.ndim != vector_order or offset.shape[0] != matrix.shape[0]:
        msg = "offset must have shape (output_dim)"
        raise ValueError(msg)
    if means.ndim != means_order:
        msg = "means must have shape (n_components, input_dim)"
        raise ValueError(msg)
    if (
        covariances.ndim != covariance_tensor_order
        or covariances.shape[-2] != covariances.shape[-1]
    ):
        msg = "covariances must have shape (n_components, input_dim, input_dim)"
        raise ValueError(msg)
    if means.shape[0] != covariances.shape[0]:
        msg = "means and covariances must have the same number of components"
        raise ValueError(msg)
    if matrix.shape[1] != means.shape[1] or matrix.shape[1] != covariances.shape[-1]:
        msg = "matrix input dimension must match means and covariance dimensions"
        raise ValueError(msg)
    if any(tensor.device != matrix.device for tensor in (offset, means, covariances)):
        msg = "matrix, offset, means, and covariances must be on the same device"
        raise ValueError(msg)

    dtype = _promoted_dtype(matrix, offset, means, covariances)
    matrix = matrix.to(dtype=dtype)
    offset = offset.to(dtype=dtype)
    means = means.to(dtype=dtype)
    covariances = covariances.to(dtype=dtype)

    transformed_means = means @ matrix.mT + offset
    transformed_covariances = torch.matmul(torch.matmul(matrix, covariances), matrix.mT)
    return transformed_means, transformed_covariances


def _promoted_dtype(*tensors: Tensor) -> torch.dtype:
    """Return the common promoted dtype for a non-empty sequence of tensors."""
    dtype = tensors[0].dtype
    for tensor in tensors[1:]:
        dtype = torch.promote_types(dtype, tensor.dtype)
    return dtype


def _validate_rotation_angle(angle: float | Tensor) -> Tensor:
    """Return a validated scalar angle tensor without changing its dtype or device."""
    angle_tensor = angle if isinstance(angle, Tensor) else torch.as_tensor(angle)
    if angle_tensor.ndim != 0:
        msg = "angle must be a scalar"
        raise ValueError(msg)
    if angle_tensor.is_complex():
        msg = "angle must be real"
        raise ValueError(msg)
    if not torch.isfinite(angle_tensor):
        msg = "angle must be finite"
        raise ValueError(msg)
    return angle_tensor


def _resolve_rotation_dtype(dtype: torch.dtype | None, *tensors: Tensor) -> torch.dtype:
    """Resolve and validate the floating-point dtype of a rotation matrix."""
    if dtype is None:
        dtype = _promoted_dtype(*tensors)
        if not dtype.is_floating_point:
            return torch.get_default_dtype()
        return dtype
    if not dtype.is_floating_point:
        msg = "rotation matrix dtype must be floating point"
        raise ValueError(msg)
    return dtype
