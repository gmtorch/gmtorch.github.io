def bayesian_fusion_gmm(
    gm0: GaussianMixture, gm1: GaussianMixture, block_size: int = 128
) -> GaussianMixture:
    """
    Fuse two Gaussian mixtures by taking the normalized product of their densities.

    This function computes the Bayesian fusion of two Gaussian mixtures
    ``gm0`` and ``gm1`` with matching dimensionality. Each pair of components
    ``(i, j)`` produces one fused Gaussian with:
    - covariance ``(Sigma0_i^{-1} + Sigma1_j^{-1})^{-1}``,
    - mean weighted by the summed precision-mean products,
    - unnormalized weight equal to the product of component weights scaled by
      the Gaussian overlap term.

    The resulting mixture has ``gm0.n_components * gm1.n_components``

    Parameters
    ----------
    gm0 : GaussianMixture
        First input Gaussian mixture.
    gm1 : GaussianMixture
        Second input Gaussian mixture. Must have the same ``dim`` as ``gm0``.
    block_size : int, default=128
        Block size used for blocked pairwise computation to control memory use.
        Must be a strictly positive integer.

    Returns
    -------
    GaussianMixture
        The fused Gaussian mixture distribution representing the normalized
        product ``p(x) ∝ p_gm0(x) * p_gm1(x)``.
    """
    if gm0.dim != gm1.dim:
        msg = "GaussianMixture dimensions must match for fusion"
        raise ValueError(msg)
    if block_size <= 0:
        msg = "block_size must be a positive integer"
        raise ValueError(msg)

    mu0, mu1 = gm0.means, gm1.means
    var0, var1 = gm0.covariances, gm1.covariances
    w0, w1 = gm0.weights, gm1.weights

    eta0_all = _precompute_precision_mean_products(var0, mu0, block_size=block_size * block_size)
    eta1_all = _precompute_precision_mean_products(var1, mu1, block_size=block_size * block_size)

    k0, d = mu0.shape
    k1 = mu1.shape[0]

    device = mu0.device
    dtype = mu0.dtype
    total = k0 * k1

    fused_weights = torch.empty(total, device=device, dtype=dtype)
    fused_means = torch.empty((total, d), device=device, dtype=dtype)
    fused_covariances = torch.empty((total, d, d), device=device, dtype=dtype)

    const = d * math.log(2.0 * math.pi)

    for i0 in range(0, k0, block_size):
        i1 = min(i0 + block_size, k0)
        bm = i1 - i0

        mu0_blk = mu0[i0:i1]
        var0_blk = var0[i0:i1]
        w0_blk = w0[i0:i1]
        eta0_blk = eta0_all[i0:i1]

        for j0 in range(0, k1, block_size):
            j1 = min(j0 + block_size, k1)
            bn = j1 - j0

            mu1_blk = mu1[j0:j1]
            var1_blk = var1[j0:j1]
            w1_blk = w1[j0:j1]
            eta1_blk = eta1_all[j0:j1]

            w_ij, mean_ij, cov_ij = _compute_fused_component_block(
                mu0_blk, mu1_blk, var0_blk, var1_blk, w0_blk, w1_blk, eta0_blk, eta1_blk, const
            )

            rows = torch.arange(bm, device=device)[:, None]
            cols = torch.arange(bn, device=device)[None, :]
            flat_idx = ((i0 + rows) * k1 + (j0 + cols)).reshape(-1)

            fused_weights[flat_idx] = w_ij.reshape(-1)
            fused_means[flat_idx] = mean_ij.reshape(-1, d)
            fused_covariances[flat_idx] = cov_ij.reshape(-1, d, d)

    z = fused_weights.sum()
    if z <= 0:
        msg = "Fusion normalization constant is non-positive"
        raise ValueError(msg)

    fused_weights = fused_weights / z

    return GaussianMixture(
        n_components=total,
        dim=d,
        weights=fused_weights,
        means=fused_means,
        covariances=fused_covariances,
        device=gm0.device,
    )




def _compute_fused_component_block(  # noqa: PLR0913
    mu0_blk: Tensor,
    mu1_blk: Tensor,
    var0_blk: Tensor,
    var1_blk: Tensor,
    w0_blk: Tensor,
    w1_blk: Tensor,
    eta0_blk: Tensor,
    eta1_blk: Tensor,
    const: float,
) -> tuple[Tensor, Tensor, Tensor]:
    """Compute fused weights, means, and covariances for one component block pair."""
    diff = mu0_blk[:, None, :] - mu1_blk[None, :, :]
    sum_var = var0_blk[:, None, :, :] + var1_blk[None, :, :, :]

    chol_sum = torch.linalg.cholesky(sum_var)
    sol_sum = torch.cholesky_solve(diff[..., None], chol_sum).squeeze(-1)
    quad = (diff * sol_sum).sum(dim=-1)
    logdet = 2.0 * torch.log(torch.diagonal(chol_sum, dim1=-2, dim2=-1)).sum(dim=-1)
    c_ij = torch.exp(-0.5 * (const + logdet + quad))

    cov_ij = _pairwise_fused_covariance(var0_blk, var1_blk)
    eta_ij = eta0_blk[:, None, :] + eta1_blk[None, :, :]
    mean_ij = (cov_ij @ eta_ij[..., None]).squeeze(-1)

    w_ij = c_ij * (w0_blk[:, None] * w1_blk[None, :])

    return w_ij, mean_ij, cov_ij


def _pairwise_fused_covariance(
    covariances_a: torch.Tensor, covariances_b: torch.Tensor, *, choose_smaller: bool = True
) -> torch.Tensor:
    """
    Compute all pairwise matrices.

        (Sigma_a[i]^{-1} + Sigma_b[j]^{-1})^{-1}

    for Sigma_a of shape [Ba, D, D] and Sigma_b of shape [Bb, D, D].

    Returns
    -------
    Sigma_ab : torch.Tensor
        Tensor of shape [Ba, Bb, D, D].
    """
    # Shapes:
    # A: [Ba, 1, D, D]  # noqa: ERA001
    # B: [1, Bb, D, D]  # noqa: ERA001
    var_a_broadcast = covariances_a[:, None, :, :]
    var_b_broadcast = covariances_b[None, :, :, :]

    # Pairwise sums: [Ba, Bb, D, D]
    sum_var = var_a_broadcast + var_b_broadcast

    if choose_smaller:
        # Frobenius norms:
        # norm_a: [Ba]  # noqa: ERA001
        # norm_b: [Bb]  # noqa: ERA001
        norms_a = torch.linalg.norm(covariances_a, ord="fro", dim=(-2, -1))
        norms_b = torch.linalg.norm(covariances_b, ord="fro", dim=(-2, -1))

        # mask: [Ba, Bb, 1, 1]  # noqa: ERA001
        use_covariance_a = (norms_a[:, None] <= norms_b[None, :])[:, :, None, None]

        # M[i, j] is either Sigma_a[i] or Sigma_b[j]
        selected_var = torch.where(use_covariance_a, var_a_broadcast, var_b_broadcast)
    else:
        # Always use the Sigma_a-side formula
        selected_var = var_a_broadcast.expand_as(sum_var)

    # Cholesky factorization of all pairwise sums
    # L[i, j] @ L[i, j].T = S[i, j]
    chol_factors = torch.linalg.cholesky(sum_var)

    # Compute Y = L^{-1} M for every pair
    transformed_var = torch.linalg.solve_triangular(chol_factors, selected_var, upper=False)

    # M - M S^{-1} M = M - (L^{-1} M)^T (L^{-1} M)
    fused_var = selected_var - transformed_var.transpose(-1, -2) @ transformed_var

    # Symmetrize to remove tiny numerical asymmetry
    fused_var = 0.5 * (fused_var + fused_var.transpose(-1, -2))

    return cast("Tensor", fused_var)


def _precompute_precision_mean_products(
    covariances: Tensor, means: Tensor, block_size: int = 128
) -> Tensor:
    """Compute Σ_k^{-1} μ_k for each component k, using blocked Cholesky solves."""
    n_components = covariances.shape[0]
    precision_mean_products = torch.empty_like(means)

    for i0 in range(0, n_components, block_size):
        i1 = min(i0 + block_size, n_components)
        chol_blk = torch.linalg.cholesky(covariances[i0:i1])
        precision_mean_products[i0:i1] = torch.cholesky_solve(
            means[i0:i1, :, None], chol_blk
        ).squeeze(-1)

    return precision_mean_products
