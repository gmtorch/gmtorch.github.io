def closed_form_propagation(
    gm: GaussianMixture, operations: Iterable[ClosedFormStep], *, max_components: int | None = None
) -> GaussianMixture:
    """Exactly propagate a Gaussian mixture through a chain of closed-form operations.

    Each operation may either implement ``ClosedFormOperation.apply`` or be a
    callable with signature ``GaussianMixture -> GaussianMixture``. The input
    mixture is cloned before the first operation, so built-in operation chains
    do not mutate the caller's mixture.

    Parameters
    ----------
    gm : GaussianMixture
        Input Gaussian mixture.
    operations : Iterable[ClosedFormStep]
        Exact propagation operations applied sequentially.
    max_components : int, optional
        Maximum allowed component count after any operation. This is useful
        because sums, differences, and fusions multiply component counts.

    Returns
    -------
    GaussianMixture
        Exact Gaussian mixture after all operations have been applied.

    Raises
    ------
    TypeError
        If an operation is neither callable nor implements ``apply``, or if an
        operation does not return a ``GaussianMixture``.
    ValueError
        If ``max_components`` is invalid or the propagated mixture exceeds it.
    """
    if max_components is not None:
        _validate_positive_integer("max_components", max_components)

    result = gm.clone()
    for operation in operations:
        result = _apply_closed_form_step(result, operation)
        if max_components is not None and result.n_components > max_components:
            msg = "closed-form propagation exceeded max_components"
            raise ValueError(msg)

    return result




def _apply_closed_form_step(gm: GaussianMixture, operation: object) -> GaussianMixture:
    """Apply one exact propagation step and validate its return type."""
    result: object
    if isinstance(operation, ClosedFormOperation):
        result = operation.apply(gm)
    elif callable(operation):
        operation_callable = cast("Callable[[GaussianMixture], object]", operation)
        result = operation_callable(gm)
    else:
        msg = "closed-form propagation operations must implement apply or be callable"
        raise TypeError(msg)

    if not isinstance(result, GaussianMixture):
        msg = "closed-form propagation operations must return a GaussianMixture"
        raise TypeError(msg)
    return result

def propagate_gmm(  # noqa: PLR0913
    gm: GaussianMixture,
    func: Callable[[Tensor], Tensor],
    n_samples: int,
    target_k: int,
    fit_kwargs: Mapping[str, object] | None = None,
    *,
    sample_seed: int | None = None,
) -> GaussianMixture:
    """Approximate the pushforward of a Gaussian mixture through ``func``.

    This function draws samples from ``gm``, applies ``func`` to the sample
    batch, and fits a new full-covariance Gaussian mixture to the transformed
    samples using ``GaussianMixtureEstimator``.

    Parameters
    ----------
    gm : GaussianMixture
        Input Gaussian mixture.
    func : Callable[[Tensor], Tensor]
        Batch function applied to samples from ``gm``. It receives a tensor of
        shape ``(n_samples, gm.dim)`` and must return either a tensor of shape
        ``(n_samples, output_dim)`` or a vector of shape ``(n_samples,)``.
    n_samples : int
        Number of samples to draw from the input mixture.
    target_k : int
        Number of components in the fitted output mixture.
    fit_kwargs : Mapping[str, object], optional
        Additional keyword arguments forwarded to ``GaussianMixtureEstimator``.
        Use ``sample_seed`` for the input-mixture sampling seed and
        ``fit_kwargs={"seed": ...}`` for the EM initialization seed.
    sample_seed : int, optional
        Seed used only when sampling from ``gm``.

    Returns
    -------
    GaussianMixture
        Gaussian-mixture approximation to ``func(X)``.

    Raises
    ------
    ValueError
        If sample counts, target component count, fit arguments, or transformed
        samples are invalid.
    """
    _validate_positive_integer("target_k", target_k)
    _validate_sample_count(n_samples, target_k)

    samples = gm.sample(n_samples, seed=sample_seed)
    with torch.no_grad():
        transformed_samples = func(samples)
    transformed_samples = _validate_transformed_samples(transformed_samples, n_samples)

    estimator = _build_estimator(target_k, fit_kwargs)
    estimator.fit(transformed_samples)
    return estimator.to_mixture()




def _build_estimator(
    target_k: int, fit_kwargs: Mapping[str, object] | None
) -> GaussianMixtureEstimator:
    """Build the EM estimator configured by propagation fit arguments."""
    estimator_kwargs = {} if fit_kwargs is None else dict(fit_kwargs)
    if "n_components" in estimator_kwargs:
        msg = "fit_kwargs must not include n_components; use target_k instead"
        raise ValueError(msg)

    try:
        return GaussianMixtureEstimator(
            n_components=target_k, **cast("dict[str, Any]", estimator_kwargs)
        )
    except TypeError as err:
        msg = "fit_kwargs contains unsupported GaussianMixtureEstimator arguments"
        raise ValueError(msg) from err


def _validate_transformed_samples(transformed_samples: object, n_samples: int) -> Tensor:
    """Validate and normalize transformed sample shape."""
    if not isinstance(transformed_samples, Tensor):
        msg = "func must return a torch.Tensor"
        raise TypeError(msg)
    if transformed_samples.ndim == 1:
        if transformed_samples.shape[0] != n_samples:
            msg = "transformed samples must have shape (n_samples,) or (n_samples, dim)"
            raise ValueError(msg)
        transformed_samples = transformed_samples.unsqueeze(1)
    elif transformed_samples.ndim == MATRIX_TENSOR_ORDER:
        if transformed_samples.shape[0] != n_samples:
            msg = "transformed samples must have shape (n_samples, dim)"
            raise ValueError(msg)
    else:
        msg = "transformed samples must have shape (n_samples,) or (n_samples, dim)"
        raise ValueError(msg)

    if transformed_samples.shape[1] <= 0:
        msg = "transformed samples must have positive dimension"
        raise ValueError(msg)
    if transformed_samples.is_complex():
        msg = "transformed samples must be real"
        raise ValueError(msg)
    if not transformed_samples.is_floating_point():
        msg = "transformed samples must be a floating point tensor"
        raise ValueError(msg)
    if not torch.isfinite(transformed_samples).all():
        msg = "transformed samples must contain only finite values"
        raise ValueError(msg)
    return transformed_samples


def _validate_sample_count(n_samples: int, target_k: int) -> None:
    """Validate propagation sample count against estimator requirements."""
    _validate_positive_integer("n_samples", n_samples)
    minimum_samples = max(2, target_k)
    if n_samples < minimum_samples:
        msg = f"n_samples must be at least {minimum_samples}"
        raise ValueError(msg)


def _validate_positive_integer(name: str, value: int) -> None:
    """Validate positive integer user input."""
    if not isinstance(value, int) or isinstance(value, bool) or value <= 0:
        msg = f"{name} must be a positive integer"
        raise ValueError(msg)
